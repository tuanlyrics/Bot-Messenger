// Gọi Gemini
async function callGemini(prompt) {
  try {
    const response = await axios.post(
      "https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=AIzaSyAIrHT8PlzLDfRwV3HAwcTapkILpH322So",
      {
        contents: [
          {
            parts: [{ text: prompt }]
          }
        ]
      },
      {
        headers: {
          "Content-Type": "application/json"
        }
      }
    );
    const result = response.data.candidates?.[0]?.content?.parts?.[0]?.text;
    return result || "Mày sủa như chó điên!";
  } catch (err) {
    console.error("Gemini error:", err?.response?.data || err.message);
    return "Lỗi gọi AI, thôi tao tự chửi: Mày ngu vãi lìn!";
  }
}



module.exports.handleEvent = async function ({ event, api }) {
  const { body, threadID, messageID, senderID, type, messageReply } = event;
  if (senderID == api.getCurrentUserID()) return;

  // Kiểm tra xem user có bị block tạm không
  const userData = spamData.get(senderID);
  const now = Date.now();

  // Nếu user bị block và chưa hết 1 phút
  if (userData?.blockedUntil && userData.blockedUntil > now) {
    return; // Bỏ qua không phản hồi
  }

  // Nếu là emoji và đúng emoji nhóm
  if (type === "message" && body && !messageReply) {
    const info = await api.getThreadInfo(threadID);
    if (body === info.emoji) {
      const insult = insultList[Math.floor(Math.random() * insultList.length)];
      return api.sendMessage(`${body} ${insult}`, threadID, messageID);
    }
  }

  // Nếu là reply vào bot
  if (messageReply && messageReply.senderID == api.getCurrentUserID()) {
    // Tăng số lần dùng
    let userSpam = spamData.get(senderID) || { count: 0, blockedUntil: 0 };
    userSpam.count += 1;

    // Nếu quá 5 lần → block 1 phút
    if (userSpam.count > 5) {
      userSpam.blockedUntil = now + 60 * 1000; // block 1 phút
      userSpam.count = 0;
      spamData.set(senderID, userSpam);
      return api.sendMessage("⚠️ Mày spam nhiều quá, đi bú đá đi", threadID, messageID);
    }

    // Lưu lại và tiếp tục xử lý
    spamData.set(senderID, userSpam);

    // Gọi Gemini AI
    const prompt = `Một người vừa nói: "${body}". Hãy chửi lại cực gắt bằng giọng chợ búa, tục tĩu, giận dữ, xúc phạm, không cần lịch sự, giống dân anh chị.`;
    const geminiReply = await callGemini(prompt);
    return api.sendMessage(`Biết cặc gì mà nói:\n${geminiReply}`, threadID, messageID);
  }
};
