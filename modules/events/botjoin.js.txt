function drawRoundedRect(ctx, x, y, width, height, radius) {
    ctx.beginPath();
    ctx.moveTo(x + radius, y);
    ctx.lineTo(x + width - radius, y);
    ctx.quadraticCurveTo(x + width, y, x + width, y + radius);
    ctx.lineTo(x + width, y + height - radius);
    ctx.quadraticCurveTo(x + width, y + height, x + width - radius, y + height);
    ctx.lineTo(x + radius, y + height);
    ctx.quadraticCurveTo(x, y + height, x, y + height - radius);
    ctx.lineTo(x, y + radius);
    ctx.quadraticCurveTo(x, y, x + radius, y);
    ctx.closePath();
}

async function getAvatarBuffer(url) {
    try {
        const res = await axios.get(url, { responseType: 'arraybuffer' });
        return Buffer.from(res.data, 'binary');
    } catch (e) {
        console.error("Lỗi khi tải avatar, sử dụng ảnh mặc định.", e.message);
        const canvas = createCanvas(200, 200);
        const ctx = canvas.getContext('2d');
        ctx.fillStyle = "#2E2E2E"; ctx.fillRect(0, 0, 200, 200);
        return canvas.toBuffer();
    }
}

// --- HÀM TẠO CANVAS THÀNH CÔNG (BỐ CỤC MỚI) ---
async function createSuccessCanvas({ botAvatarUrl, groupName, botName, prefix }) {
    const width = 1200, height = 500;
    const canvas = createCanvas(width, height);
    const ctx = canvas.getContext('2d');
    const cacheDir = path.join(__dirname, '../../cache');

    // 1. Vẽ Background
    try {
        const bgFiles = fs.readdirSync(cacheDir).filter(f => f.startsWith('bg_join_'));
        const randomBgPath = path.join(cacheDir, bgFiles[Math.floor(Math.random() * bgFiles.length)]);
        const bgImage = await loadImage(randomBgPath);
        const imgRatio = bgImage.width / bgImage.height;
        const canvasRatio = width / height;
        let sx = 0, sy = 0, sWidth = bgImage.width, sHeight = bgImage.height;
        if (imgRatio > canvasRatio) { 
            sWidth = sHeight * canvasRatio; sx = (bgImage.width - sWidth) / 2; 
        } else { 
            sHeight = sWidth / canvasRatio; sy = (bgImage.height - sHeight) / 2; 
        }
        ctx.drawImage(bgImage, sx, sy, sWidth, sHeight, 0, 0, width, height);
    } catch (e) {
        ctx.fillStyle = '#2c2c2c'; ctx.fillRect(0, 0, width, height);
    }

    // 2. Hiệu ứng "Liquid Glass"
    ctx.save();
    const padding = 50, boxX = padding, boxY = padding;
    const boxWidth = width - padding * 2, boxHeight = height - padding * 2;
    drawRoundedRect(ctx, boxX, boxY, boxWidth, boxHeight, 35);
    ctx.fillStyle = "rgba(0, 0, 0, 0.45)";
    ctx.shadowColor = "rgba(0, 0, 0, 0.6)";
    ctx.shadowBlur = 30;
    ctx.fill();
    ctx.shadowColor = "transparent";
    ctx.strokeStyle = "rgba(255, 255, 255, 0.2)";
    ctx.lineWidth = 1.5;
    ctx.stroke();
    ctx.restore();

    // 3. Avatar Bot
    const avatarSize = 180;
    const avatarX = boxX + 50;
    const avatarY = height / 2 - avatarSize / 2;
    ctx.save();
    ctx.beginPath();
    ctx.arc(avatarX + avatarSize / 2, avatarY + avatarSize / 2, avatarSize / 2 + 5, 0, Math.PI * 2);
    ctx.fillStyle = '#FFFFFF';
    ctx.shadowColor = "#32ff7e";
    ctx.shadowBlur = 40;
    ctx.fill();
    ctx.restore();
    ctx.save();
    ctx.beginPath();
    ctx.arc(avatarX + avatarSize / 2, avatarY + avatarSize / 2, avatarSize / 2, 0, Math.PI * 2);
    ctx.clip();
    const avatar = await loadImage(await getAvatarBuffer(botAvatarUrl));
    ctx.drawImage(avatar, avatarX, avatarY, avatarSize, avatarSize);
    ctx.strokeStyle = "rgba(255, 255, 255, 0.8)";
    ctx.lineWidth = 4;
    ctx.stroke();
    ctx.restore();

    // 4. Typography
    const textAreaX = avatarX + avatarSize + 50;
    let currentY = boxY + 100;
    
// === TITLE (RAINBOW GRADIENT + GLOW) ===
ctx.font = 'bold 40px "Signika-SemiBold"';
ctx.textBaseline = 'top';

const titleText = "KẾT NỐI THÀNH CÔNG";
const titleMetrics = ctx.measureText(titleText);
const titleWidth = titleMetrics.width;

// Căn giữa phần text bên phải avatar, rồi xích trái 20px
const titleX = textAreaX + ((boxWidth - avatarSize - 100 - titleWidth) / 2) - 120;
const titleY = boxY + 40; // Xích lên 10px so với cũ

// Gradient cầu vồng
const grad = ctx.createLinearGradient(titleX, 0, titleX + titleWidth, 0);
const rainbowColors = ["#ff3838", "#ff9f1a", "#fff200", "#32ff7e", "#18dcff", "#7d5fff", "#f368e0"];
rainbowColors.forEach((color, i) => {
    grad.addColorStop(i / (rainbowColors.length - 1), color);
});

ctx.fillStyle = grad;
ctx.shadowColor = "rgba(255, 255, 255, 0.8)";
ctx.shadowBlur = 40;
ctx.fillText(titleText, titleX, titleY);
ctx.shadowColor = "transparent";

// Cập nhật vị trí để vẽ tiếp nội dung dưới
currentY = titleY + 60;

    // Info
    ctx.font = '30px "Signika-SemiBold"';
    ctx.fillStyle = '#FFFFFF';
    ctx.fillText(`Bot: ${botName}`, textAreaX, currentY);
    currentY += 45;
    ctx.fillText(`Nhóm: ${groupName}`, textAreaX, currentY);
    currentY += 45;
    ctx.fillText(`Prefix: ${prefix}`, textAreaX, currentY);
    currentY += 45;
    
    // Time
    const time = moment.tz("Asia/Ho_Chi_Minh").format('HH:mm:ss - DD/MM/YYYY');
    ctx.font = '26px "Signika-SemiBold"';
    ctx.fillStyle = '#DDDDDD';
    ctx.fillText(`Thời gian: ${time}`, textAreaX, currentY);
    currentY += 65;

    // Final message
    ctx.font = 'italic 24px "Signika-SemiBold"';
    ctx.fillStyle = '#CCCCCC';
    ctx.fillText("Chúc mọi người có những trải nghiệm vui vẻ!", textAreaX, currentY);
    
    return canvas.toBuffer('image/png');
}

        api.sendMessage("Bot đang khởi tạo, vui lòng chờ trong giây lát...", threadID);

        // Đợi 5 giây
        setTimeout(async () => {
            try {
                const { threadName } = await Threads.getInfo(threadID);
                const threadSetting = global.data.threadData.get(threadID) || {};
                const prefix = threadSetting.PREFIX || PREFIX;
                const botID = api.getCurrentUserID();
                const botAvatarUrl = `https://graph.facebook.com/${botID}/picture?width=720&height=720&access_token=6628568379%7Cc1e620fa708a1d5696fb991c1bde5662`;

                const successCanvasBuffer = await createSuccessCanvas({
                    botAvatarUrl,
                    groupName: threadName,
                    botName: BOTNAME,
                    prefix: prefix
                });
                
                const pngPath = path.join(cacheDir, `success_${threadID}.png`);
                fs.writeFileSync(pngPath, successCanvasBuffer);
                
                const msg = {
    body: `🌐 Kết nối thành công!
──────────
👉 Dùng "${prefix}rent thanhtoan" để thuê bot
👉 Dùng "${prefix}menu" để xem danh sách lệnh
──────────
❗ Mọi thắc mắc hoặc báo lỗi, vui lòng liên hệ Admin.
──────────`,
    attachment: fs.createReadStream(pngPath)
};


            api.sendMessage(msg, threadID, (err, info) => {
                 if (err) return console.error(err);
                 fs.unlinkSync(pngPath);
                 if (ADMINBOT && ADMINBOT.length > 0) {
                     api.shareContact("──────────\n🌐 Hệ thống tính điểm: https://spn.io.vn\n🤝 Thuê bot: liên hệ trực tiếp Nguyễn Phát\n──────────", ADMINBOT[0], threadID);
                 }
            });

            } catch (e) {
                console.error("Lỗi trong sự kiện botJoin:", e);
                api.sendMessage(`Đã xảy ra lỗi khi tạo ảnh thông báo, nhưng bot đã kết nối thành công! Vui lòng dùng "${PREFIX}help" để bắt đầu.`, threadID);
            }
        }, 2000); // Độ trễ 5 giây
    }
